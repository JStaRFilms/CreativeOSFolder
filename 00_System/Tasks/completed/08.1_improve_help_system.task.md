```python
  cos sync --pull-only              Only copy from vault (one-way)
  cos sync -v                       Verbose output with file list

Output Legend:
  [green]PUSH[/green]  - File copied from project to vault
  [blue]PULL[/blue]   - File copied from vault to project
  [dim]SKIP[/dim]     - Files are identical, no action
        """,
        formatter_class=argparse.RawDescriptionHelpFormatter
    )
    
    p_sync.add_argument(
        "--project",
        type=str,
        help="Sync only the specified project (by name or slug)"
```

```python
"""Custom help formatter using Rich."""

from argparse import HelpFormatter, Action
from rich.console import Console
from rich.panel import Panel
from rich.table import Table
from rich.text import Text
import re

class RichHelpFormatter(HelpFormatter):
    """Argparse help formatter that uses Rich for beautiful output."""
    
    def __init__(self, prog, indent_increment=2, max_help_position=24, width=None):
        super().__init__(prog, indent_increment, max_help_position, width)
        self.console = Console()
    
    def format_help(self):
        """Format the help message with Rich."""
        # Get the standard help text
        help_text = super().format_help()
        
        # Parse and reformat with Rich
        return self._beautify_help(help_text)
    
    def _beautify_help(self, text):
        """Convert plain help text to Rich-formatted output."""
        # This is called by argparse when help is requested
        return text


class RichHelpAction(Action):
    """Custom action for --help that uses Rich formatting."""
    
    def __init__(self, option_strings, dest=argparse.SUPPRESS, default=argparse.SUPPRESS, help=None):
        super().__init__(option_strings=option_strings, dest=dest, default=default, nargs=0, help=help)
    
    def __call__(self, parser, namespace, values, option_string=None):
        """Print richly formatted help and exit."""
        console = Console()
        
        # Print description
        if parser.description:
            console.print(Panel(parser.description, title="Description", border_style="cyan"))
        
        # Print usage
        console.print(f"\n[bold]Usage:[/bold] {parser.prog} [options]\n")
        
        # Print options in a table
        options_table = Table(title="Options", box=box.ROUNDED)
        options_table.add_column("Flag", style="cyan bold")
        options_table.add_column("Description", style="white")
        
        for action in parser._actions:
            if action.help and action.option_strings:
                flags = ", ".join(action.option_strings)
                options_table.add_row(flags, action.help or "")
        
        console.print(options_table)
        
        # Print examples if available
        if parser.epilog:
            console.print(Panel(parser.epilog, title="Examples", border_style="green"))
        
        parser.exit()
```
