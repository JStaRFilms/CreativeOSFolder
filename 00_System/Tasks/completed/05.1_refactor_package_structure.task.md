### `cos/cli.py`

```python
"""Main CLI entry point."""

import argparse
import sys
from . import __version__
from .console import console
from .commands import new, clone, init, sync, export, thumbs, clean, sort_exports, travel, resurrect

def main():
    """Main entry point for CreativeOS CLI."""
    parser = argparse.ArgumentParser(
        description="CreativeOS - The Central Nervous System for Creative Workflows",
        formatter_class=argparse.RawTextHelpFormatter
    )
    subparsers = parser.add_subparsers(dest="command", title="Commands")
    
    # Add subparsers for each command
    new.add_parser(subparsers)
    clone.add_parser(subparsers)
    # ... etc.
    
    args = parser.parse_args()
    
    # Route to command handler
    if args.command == "new":
        new.cmd_new(args)
    elif args.command == "clone":
        clone.cmd_clone(args)
    # ... etc.


if __name__ == "__main__":
    main()
```
