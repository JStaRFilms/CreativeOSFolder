### 2. Create `00_System/Scripts/cos/category_config.py`

```python
"""Category configuration loading and management."""

import os
import json
from typing import Dict, List, Optional, Any
from dataclasses import dataclass

from .console import console

# Path to categories.json (relative to this file)
_CATEGORY_CONFIG_PATH = os.path.join(
    os.path.dirname(os.path.dirname(os.path.dirname(os.path.abspath(__file__)))),
    "Config", "categories.json"
)

@dataclass
class CategoryConfig:
    """Configuration for a single category."""
    name: str
    template: str
    physical_folder: str
    aliases: List[str]
    description: str


class CategoryManager:
    """Manages category configuration loaded from JSON."""
    
    _instance: Optional['CategoryManager'] = None
    _categories: Dict[str, CategoryConfig] = {}
    _default_category: str = "Video"
    _simple_template: str = "simple"
    
    def __new__(cls) -> 'CategoryManager':
        if cls._instance is None:
            cls._instance = super().__new__(cls)
            cls._instance._load_config()
        return cls._instance
    
    def _load_config(self) -> None:
        """Load categories from JSON file."""
        if not os.path.exists(_CATEGORY_CONFIG_PATH):
            console.print("[warning]⚠️  categories.json not found, using defaults[/warning]")
            self._load_defaults()
            return
        
        try:
            with open(_CATEGORY_CONFIG_PATH, "r", encoding="utf-8") as f:
                config = json.load(f)
            
            self._default_category = config.get("default_category", "Video")
            self._simple_template = config.get("simple_template", "simple")
            
            for name, cat_config in config.get("categories", {}).items():
                self._categories[name] = CategoryConfig(
                    name=name,
                    template=cat_config.get("template", f"{name.lower()}_project"),
                    physical_folder=cat_config.get("physical_folder", name),
                    aliases=cat_config.get("aliases", []),
                    description=cat_config.get("description", "")
                )
                
        except (json.JSONDecodeError, IOError) as e:
            console.print(f"[error]❌ Error loading categories.json: {e}[/error]")
            self._load_defaults()
    
    def _load_defaults(self) -> None:
        """Load minimal default categories if config is missing."""
        defaults = ["Video", "Code", "Audio", "AI", "Design", "Photo", "Writing"]
        for cat in defaults:
            self._categories[cat] = CategoryConfig(
                name=cat,
                template=f"{cat.lower()}_project",
                physical_folder=cat,
                aliases=[],
                description=f"{cat} projects"
            )
    
    def get_all_categories(self) -> List[str]:
        """Get list of all valid category names (including aliases)."""
        names = list(self._categories.keys())
        for cat in self._categories.values():
            names.extend(cat.aliases)
        return sorted(set(names))
    
    def get_primary_categories(self) -> List[str]:
        """Get list of primary category names (no aliases)."""
        return sorted(self._categories.keys())
    
    def resolve_category(self, name: str) -> str:
        """Resolve a category name (handling aliases) to primary name."""
        name_lower = name.lower()
        
        # Check direct match
        if name in self._categories:
            return name
        
        # Check case-insensitive match
        for cat_name in self._categories:
            if cat_name.lower() == name_lower:
                return cat_name
        
        # Check aliases
        for cat_name, cat_config in self._categories.items():
            if name_lower in [a.lower() for a in cat_config.aliases]:
                return cat_name
        
        # Return default if not found
        return self._default_category
    
    def get_template(self, category: str) -> str:
        """Get template name for a category."""
        resolved = self.resolve_category(category)
        if resolved in self._categories:
            return self._categories[resolved].template
        return f"{category.lower()}_project"
    
    def get_physical_folder(self, category: str) -> str:
        """Get physical folder name for a category."""
        resolved = self.resolve_category(category)
        if resolved in self._categories:
            return self._categories[resolved].physical_folder
        return category
    
    def get_simple_template(self) -> str:
        """Get the simple template name."""
        return self._simple_template
    
    def get_default_category(self) -> str:
        """Get the default category name."""
        return self._default_category
    
    def is_valid_category(self, name: str) -> bool:
        """Check if a category name is valid (including aliases)."""
        return name in self.get_all_categories()


# Module-level convenience functions
_manager: Optional[CategoryManager] = None

def get_manager() -> CategoryManager:
    """Get the singleton CategoryManager instance."""
    global _manager
    if _manager is None:
        _manager = CategoryManager()
    return _manager

def get_all_categories() -> List[str]:
    """Get all valid category names."""
    return get_manager().get_all_categories()

def resolve_category(name: str) -> str:
    """Resolve a category name to its primary form."""
    return get_manager().resolve_category(name)

def get_template(category: str) -> str:
    """Get template for a category."""
    return get_manager().get_template(category)

def get_physical_folder(category: str) -> str:
    """Get physical folder for a category."""
    return get_manager().get_physical_folder(category)
```
